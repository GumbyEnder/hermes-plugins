### Task 2.1: Time-Series Sparklines

**Objective:** Mini SVG line charts in each stat card showing 24h trend.

**Approach:**
- Store rolling buffer of last 50 stats snapshots (30s polls × 25 min window)
- Render 80×24px SVG polyline in each StatCard
- Color: green (trending up), red (trending down), grey (flat)

**Files:**
- Modify: `dashboard/dist/index.js` — add `sparklineData` ref, SVG rendering
- Modify: `dashboard/plugin.css` — sparkline styling

**Implementation:**
```js
// Add at top of HonchoMemory():
var sparklineRef = React.useRef({ documents: [], messages: [], embeddings: [], queue_pending: [] });

// In fetchStats callback, after setStats:
var sl = sparklineRef.current;
Object.keys(sl).forEach(function(k) {
  sl[k].push(data[k]);
  if (sl[k].length > 50) sl[k].shift();
});

// Helper function:
function Sparkline(_a) {
  var data = _a.data, width = _a.width || 80, height = _a.height || 24;
  if (!data || data.length < 2) return null;
  var min = Math.min.apply(null, data);
  var max = Math.max.apply(null, data);
  var range = max - min || 1;
  var points = data.map(function(v, i) {
    var x = (i / (data.length - 1)) * width;
    var y = height - ((v - min) / range) * (height - 2) - 1;
    return x.toFixed(1) + "," + y.toFixed(1);
  }).join(" ");
  var trend = data[data.length-1] > data[0] ? "#4ade80" : data[data.length-1] < data[0] ? "#f87171" : "#9ca3af";
  return React.createElement("svg", { width: width, height: height, className: "sparkline" },
    React.createElement("polyline", { points: points, fill: "none", stroke: trend, strokeWidth: 1.5 })
  );
}

// Add <Sparkline data={sparklineRef.current.documents} /> inside each StatCard
```

### Task 2.3: CSV Export

**Objective:** Download button that exports stats + per-agent data as CSV.

```js
// Helper function
function exportCSV(stats) {
  var rows = [
    ["Metric", "Value"],
    ["Documents", stats.documents],
    ["Messages", stats.messages],
    ["Embeddings", stats.embeddings],
    ["Documents (24h)", stats.documents_24h],
    ["Queue Pending", stats.queue_pending],
    ["Healthy", stats.healthy],
    [""],
    ["Agent", "Documents"]
  ];
  if (stats.peers) {
    Object.entries(stats.peers).forEach(function(_a) {
      rows.push([_a[0], _a[1]]);
    });
  }
  var csv = rows.map(function(r) { return r.join(","); }).join("\n");
  var blob = new Blob([csv], { type: "text/csv" });
  var url = URL.createObjectURL(blob);
  var a = document.createElement("a");
  a.href = url;
  a.download = "honcho-memory-" + new Date().toISOString().slice(0,10) + ".csv";
  a.click();
  URL.revokeObjectURL(url);
}

// Add button in the header:
React.createElement("button", { onClick: function() { return exportCSV(stats); }, className: "export-btn" }, "Export CSV")
```

function timeAgo(isoString) {
  if (!isoString) return "never";
  var diff = (Date.now() - new Date(isoString).getTime()) / 1000;
  if (diff < 60) return "just now";
  if (diff < 3600) return Math.floor(diff/60) + "m ago";
  if (diff < 86400) return Math.floor(diff/3600) + "h ago";
  return Math.floor(diff/86400) + "d ago";
}
